function fmt(v) {
  if (v === null || v === undefined) return "–";
  if (typeof v !== "number") return String(v);
  if (Number.isInteger(v)) return v.toLocaleString("ko-KR");
  return (Math.round(v * 10) / 10).toLocaleString("ko-KR", { minimumFractionDigits: 1, maximumFractionDigits: 1 });
}
function esc(s) { return String(s).replace(/[&<>"]/g, c => ({ "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;" }[c])); }
function gc(id) { return `var(--g-${id})`; }
function gbg(id) { return `var(--g-${id}-bg)`; }

/* ── filtering ── */
function taskMatchBase(t) {
  if (state.goal && t.goal_id !== state.goal) return false;
  if (state.spec && !t.특성화연계) return false;
  if (state.q) {
    const q = state.q.toLowerCase();
    const hay = [t.code, t.name, t.책임부서, ...(t.연관부서 || []),
      ...(t.subtasks || []).map(s => s.name),
      ...(t.kpi_codes || []).map(c => (kpiByCode[c] || {}).name || "")].join(" ").toLowerCase();
    if (!hay.includes(q)) return false;
  }
  return true;
}
function taskMatch(t) {
  if (state.dept && t.책임부서 !== state.dept) return false;
  return taskMatchBase(t);
}
function kpiMatch(k) {
  const t = taskByShort[k.task_code];
  if (state.goal && k.goal_id !== state.goal) return false;
  if (state.dept && (!t || t.책임부서 !== state.dept)) return false;
  if (state.spec && (!t || !t.특성화연계)) return false;
  if (state.q) {
    const q = state.q.toLowerCase();
    const hay = [k.code, k.name, k.formula || "", t ? t.name : "", t ? t.책임부서 : ""].join(" ").toLowerCase();
    if (!hay.includes(q)) return false;
  }
  return true;
}

/* ── KPI view ── */
function sparkline(k) {
  const vals = YEARS.map(y => k.targets ? k.targets[y] : null);
  const nums = vals.filter(v => typeof v === "number");
  if (typeof k.baseline === "number") nums.push(k.baseline);
  if (!nums.length) return `<span class="null">–</span>`;
  let lo = Math.min(...nums), hi = Math.max(...nums);
  if (hi === lo) { hi += 1; lo -= 1; }
  const W = 132, H = 30, P = 4;
  const x = i => P + i * (W - 2 * P) / (YEARS.length - 1);
  const y = v => H - P - (v - lo) * (H - 2 * P) / (hi - lo);
  let d = "", started = false;
  vals.forEach((v, i) => {
    if (typeof v !== "number") { started = false; return; }
    d += (started ? "L" : "M") + x(i).toFixed(1) + " " + y(v).toFixed(1) + " ";
    started = true;
  });
  const col = gc(k.goal_id);
  let lastIdx = -1;
  vals.forEach((v, i) => { if (typeof v === "number") lastIdx = i; });
  const pts = vals.map((v, i) => {
    if (typeof v !== "number") return "";
    const end = i === lastIdx;
    return `<circle class="pt" cx="${x(i).toFixed(1)}" cy="${y(v).toFixed(1)}" r="${end ? 3.4 : 2.2}"
      fill="${end ? col : "var(--card)"}" stroke="${col}" stroke-width="1.6"
      data-k="${esc(k.code)}" data-y="${YEARS[i]}"></circle>`;
  }).join("");
  const base = (typeof k.baseline === "number")
    ? `<line x1="${P}" x2="${W - P}" y1="${y(k.baseline).toFixed(1)}" y2="${y(k.baseline).toFixed(1)}"
        stroke="var(--ink-3)" stroke-width="1" stroke-dasharray="2 3" opacity=".7"></line>` : "";
  return `<svg class="spark" width="${W}" height="${H}" viewBox="0 0 ${W} ${H}" data-k="${esc(k.code)}" role="img"
    aria-label="${esc(k.name)} 목표 궤적">${base}<path d="${d.trim()}" fill="none" stroke="${col}" stroke-width="2"
    stroke-linecap="round" stroke-linejoin="round"></path>${pts}</svg>`;
}
const SORT_KEYS = {
  code: k => k.code,
  baseline: k => (typeof k.baseline === "number" ? k.baseline : null),
  t27: k => (k.targets && typeof k.targets["2027"] === "number" ? k.targets["2027"] : null)
};
function sortKpis(list) {
  const { key, dir } = state.sort;
  if (!key || !SORT_KEYS[key]) return list;
  const get = SORT_KEYS[key];
  return [...list].sort((a, b) => {
    const va = get(a), vb = get(b);
    const na = va === null || va === undefined, nb = vb === null || vb === undefined;
    if (na && nb) return 0;
    if (na) return 1;            /* null은 방향과 무관하게 항상 끝 */
    if (nb) return -1;
    if (typeof va === "string") return va.localeCompare(vb) * dir;
    return (va - vb) * dir;
  });
}

  const tys = YEARS.map(y => {

    <div class="kd-grid">
      <div class="kd wide"><h5>산출식</h5><p>${esc(k.formula || "–")}</p></div>
      <div class="kd"><h5>소속 실행과제</h5><p>${t ? esc(t.code) + " · " + esc(t.name) : "–"}</p></div>
      <div class="kd"><h5>책임부서</h5><p>${t ? esc(t.책임부서) : "–"}</p></div>

      <div class="kd wide"><h5>연도별 목표 (2022~2027)</h5><div class="kd-targets">${tys}</div></div>

}
function renderKpi() {
  const root = document.getElementById("view-kpi");
  const list = sortKpis(DATA.kpis.filter(kpiMatch));
  if (!list.length) { root.innerHTML = `<div class="empty">조건에 맞는 KPI가 없습니다.</div>`; return 0; }
  const rows = list.map(k => {
    const t = taskByShort[k.task_code];
    const open = openKpis.has(k.code);
    return `<tr class="krow${open ? " kopen" : ""}" data-code="${esc(k.code)}" tabindex="0"
      aria-expanded="${open ? "true" : "false"}" title="클릭하면 산출식·출처·연도별 목표 상세를 펼칩니다">
      <td><span class="kcode"><span class="caret">▶</span><span class="dot" style="background:${gc(k.goal_id)}"></span>${esc(k.code)}</span></td>
      <td><span class="kname">${esc(k.name)}</span>
        <span class="ktask">${t ? esc(t.name) + " · " + esc(t.책임부서) : ""}</span></td>
      <td><span class="unit">${esc(k.unit || "")}</span></td>
      <td class="num">${fmt(k.baseline)}</td>
      ${YEARS.map(y => `<td class="num">${fmt(k.targets ? k.targets[y] : null)}</td>`).join("")}
      <td>${sparkline(k)}</td>
    </tr>${open ? kpiDetailRow(k) : ""}`;
  }).join("");
  const arr = key => state.sort.key === key ? `<span class="arr">${state.sort.dir === 1 ? "▲" : "▼"}</span>` : "";
  root.innerHTML = `<div class="tablewrap"><table>
    <thead><tr>
      <th class="sortable" data-sort="code" title="코드로 정렬">코드${arr("code")}</th>
      <th>지표명 · 실행과제 · 책임부서</th><th>단위</th>
      <th class="num sortable" data-sort="baseline" title="기준값으로 정렬">기준값${arr("baseline")}</th>
      ${YEARS.map(y => {
        const short = "'" + y.slice(2) + " 목표";
        return y === "2027"
          ? `<th class="num sortable" data-sort="t27" title="2027 목표로 정렬">${short}${arr("t27")}</th>`
          : `<th class="num">${short}</th>`;
      }).join("")}
      <th>목표 궤적 '22–'27 <span style="font-weight:500;text-transform:none;letter-spacing:0">(점선 = 기준값)</span></th>
    </tr></thead><tbody>${rows}</tbody></table></div>`;
  root.querySelectorAll("th.sortable").forEach(th => th.addEventListener("click", () => {
    const key = th.dataset.sort;
    if (state.sort.key === key) state.sort.dir = -state.sort.dir;
    else { state.sort.key = key; state.sort.dir = 1; }
    renderAll();
  }));
  root.querySelectorAll("tr.krow").forEach(tr => {
    const toggle = () => {
      const c = tr.dataset.code;
      if (openKpis.has(c)) openKpis.delete(c); else openKpis.add(c);
      renderAll();
    };
    tr.addEventListener("click", toggle);
    tr.addEventListener("keydown", e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); toggle(); } });
  });
  return list.length;
}

/* ── dept view ── */
function renderDept() {
  const root = document.getElementById("view-dept");
  const ftasks = DATA.tasks.filter(taskMatchBase);   /* 부서별 뷰: 부서 필터 제외, 전략·검색·특성화만 적용 */
  const map = {};
  ftasks.forEach(t => {
    if (!map[t.책임부서]) map[t.책임부서] = { total: 0, byGoal: {}, kpi: 0, tasks: [] };
    map[t.책임부서].total++;
    map[t.책임부서].byGoal[t.goal_id] = (map[t.책임부서].byGoal[t.goal_id] || 0) + 1;
    map[t.책임부서].kpi += (t.kpi_codes || []).length;
    map[t.책임부서].tasks.push(t);
  });
  const entries = Object.entries(map).sort((a, b) => b[1].total - a[1].total || a[0].localeCompare(b[0], "ko"));
  if (!entries.length) { root.innerHTML = `<div class="empty">조건에 맞는 부서가 없습니다.</div>`; return 0; }
  const max = Math.max(...entries.map(([, v]) => v.total));
  const legend = `<div class="legend">` + GOALS.map(g =>
    `<span><span class="dot" style="background:${gc(g.id)}"></span>${g.id}. ${esc(g.name)}</span>`).join("") + `</div>`;
  const bars = entries.map(([d, v]) => {
    const segs = GOALS.map(g => {
      const n = v.byGoal[g.id] || 0;
      if (!n) return "";
      const w = (n / max * 100).toFixed(2);
      return `<span class="seg" style="width:${w}%;background:${gc(g.id)}" data-tip="${esc(d)} · ${g.id} ${esc(g.name)}: ${n}건"></span>`;
    }).join("");
    const open = openDepts.has(d);
    const rel = ftasks.filter(t => (t.연관부서 || []).includes(d)).length;
    const tlist = v.tasks.map(t =>
      `<li><span class="b sid" style="--gc:${gc(t.goal_id)};--gc-bg:${gbg(t.goal_id)}">${esc(t.strategy_id)}</span>
        <span class="b">${esc(t.code)}</span><span>${esc(t.name)}</span></li>`).join("");
    return `<div class="deptrow${open ? " open" : ""}" data-dept="${esc(d)}">
      <button class="dbar" data-dept="${esc(d)}" aria-expanded="${open ? "true" : "false"}"
        title="클릭하면 이 부서의 책임 실행과제 목록을 펼칩니다">
        <span class="dn">${esc(d)}<small>연계 KPI ${v.kpi}개</small></span>
        <span class="track">${segs}</span>
        <span class="tot">${v.total}</span>
      </button>
      <div class="dpanel">
        <h4>책임 실행과제 ${v.total}건</h4>

function fmt1(v) {
  if (v === null || v === undefined || typeof v !== "number") return "–";
  return (Math.round(v * 10) / 10).toLocaleString("ko-KR", { minimumFractionDigits: 1, maximumFractionDigits: 1 });
}

function cmpChart(ind, series) {
  const W = 338, H = 132, L = 40, R = 232, T = 12, B = 110;
  const nums = series.flatMap(s => s.vals).filter(v => typeof v === "number");
  if (!nums.length) return `<div class="empty">–</div>`;
  let lo = Math.min(...nums), hi = Math.max(...nums);
  const pad = (hi - lo) * 0.12 || 1;
  lo -= pad; hi += pad;
  const x = i => L + i * (R - L) / (CMP.years.length - 1);
  const y = v => B - (v - lo) * (B - T) / (hi - lo);
  const grid = [0, 0.5, 1].map(f => {
    const v = lo + (hi - lo) * f, yy = y(v).toFixed(1);
    return `<line x1="${L}" x2="${R}" y1="${yy}" y2="${yy}" stroke="var(--line-soft)" stroke-width="1"></line>
      <text x="${L - 5}" y="${(+yy) + 3.5}" text-anchor="end" font-size="9.5" fill="var(--ink-3)">${fmt1(v)}</text>`;
  }).join("");
  const xlab = CMP.years.map((yr, i) =>
    `<text x="${x(i).toFixed(1)}" y="${H - 6}" text-anchor="middle" font-size="10" fill="var(--ink-3)">${yr}</text>`).join("");
  const lines = series.map(s => {
    let d = "", started = false;
    s.vals.forEach((v, i) => {
      if (typeof v !== "number") { started = false; return; }
      d += (started ? "L" : "M") + x(i).toFixed(1) + " " + y(v).toFixed(1) + " ";
      started = true;
    });
    const pts = s.vals.map((v, i) => (typeof v !== "number") ? "" :
      `<circle cx="${x(i).toFixed(1)}" cy="${y(v).toFixed(1)}" r="${s.em ? 3.6 : 2.8}" fill="${s.em ? s.col : "var(--card)"}"
        stroke="${s.col}" stroke-width="1.8" data-tip="${esc(ind.name)} · ${esc(s.label)} · ${CMP.years[i]} 공시: ${fmt1(v)}%"></circle>`).join("");
    let li = -1;
    s.vals.forEach((v, i) => { if (typeof v === "number") li = i; });
    const dl = li >= 0 ? `<text x="${R + 7}" y="${(y(s.vals[li]) + 3.5).toFixed(1)}" font-size="10"
      font-weight="${s.em ? 700 : 500}" fill="${s.col}">${esc(s.label)}</text>` : "";
    return `<path d="${d.trim()}" fill="none" stroke="${s.col}" stroke-width="${s.em ? 2.4 : 1.7}"
      stroke-linecap="round" stroke-linejoin="round"${s.em ? "" : ` opacity=".85"`}></path>${pts}${dl}`;
  }).join("");
  return `<svg class="cmpsvg" viewBox="0 0 ${W} ${H}" role="img" aria-label="${esc(ind.name)} 3개년 비교">${grid}${xlab}${lines}</svg>`;
}

function evalRec(code, year) {
  const y = year || state.evalYear;
  if (!evalState[y]) evalState[y] = {};
  if (!evalState[y][code]) {
    const rec = { dept: {}, ir: {}, kpi: {} };
    EVAL_SECTIONS.forEach(s => s.fields.forEach(f => { if (f.def) rec[s.id][f.key] = f.def; }));
    evalState[y][code] = rec;
  }
  return evalState[y][code];
}
function evalVal(code, sec, key) {
  const v = evalRec(code)[sec][key];
  return (v === null || v === undefined) ? "" : String(v);
}
function evalFilled(code) {
  const r = evalRec(code);
  let n = 0;
  EVAL_SECTIONS.forEach(s => s.fields.forEach(f => {
    const v = String(r[s.id][f.key] || "").trim();
    if (v !== "" && v !== (f.def || "")) n++;   /* 기본 선택값(IR 보통)은 '작성'으로 세지 않음 */
  }));
  return n;
}
/* 작성 완료 기준: 부서 자체점검 + IR센터 평가 등급이 모두 선택되고, 부서·IR 각각 서술 1개 이상 작성.
   미작성 = 평가 필드·KPI 실적값 어느 것도 입력되지 않은 상태(기본 선택값만 있는 상태 포함). 그 외는 부분작성. */
function evalStatus(code) {
  const r = evalRec(code);
  const hasText = sec => EVAL_FIELDS[sec].some(f =>
    f.type === "textarea" && String(r[sec][f.key] || "").trim() !== "");
  if (String(r.dept.grade || "") !== "" && String(r.ir.grade || "") !== "" && hasText("dept") && hasText("ir")) return "done";
  const anyKpi = Object.keys(r.kpi).some(k => String(r.kpi[k] || "").trim() !== "");
  return (evalFilled(code) > 0 || anyKpi) ? "part" : "none";
}
/* 달성률(%) = 실적 ÷ 목표 × 100. 목표 공란(null)·0 또는 실적 미입력이면 null → 화면에 "–" */
function evalRate(actual, target) {
  const s = String(actual === null || actual === undefined ? "" : actual).trim();
  if (s === "") return null;
  const a = parseFloat(s);
  if (!isFinite(a)) return null;
  if (typeof target !== "number" || !isFinite(target) || target === 0) return null;
  return a / target * 100;
}
function evalSummaryInner(list) {
  let done = 0, part = 0;
  list.forEach(t => { const s = evalStatus(t.code); if (s === "done") done++; else if (s === "part") part++; });
  const none = list.length - done - part;
  return `<span>실행과제 <b>${list.length}</b>건</span>` +
    `<span class="sdone">작성 완료 <b>${done}</b>건</span>` +
    `<span class="spart">부분작성 <b>${part}</b>건</span>` +
    `<span class="snone">미작성 <b>${none}</b>건</span>`;
}

}

    const card = b.closest(".evcard"), c = card.dataset.code;

    b.setAttribute("aria-expanded", open ? "true" : "false");
    if (open) { openEvals.add(c); card.querySelectorAll("textarea").forEach(autoGrow); }
    else openEvals.delete(c);
  }));
  /* 입력 시 전체 재렌더를 하지 않는다(포커스·커서 보존). evalState 갱신 후 관련 표시만 부분 갱신. */
  root.querySelectorAll("[data-ev='f']").forEach(el => {
    const onEdit = () => {
      if (!el.isConnected) return;   /* 재렌더로 떨어져 나간 낡은 입력칸의 지연 이벤트는 무시 */
      evalRec(el.dataset.code, el.dataset.year)[el.dataset.sec][el.dataset.key] = el.value;
      const mirror = el.parentElement.querySelector(".ev-print");
      if (mirror) mirror.textContent = el.value;
      if (el.tagName === "TEXTAREA") autoGrow(el);
      refreshEvalMeta(el.dataset.code);
    };
    el.addEventListener("input", onEdit);
    el.addEventListener("change", onEdit);
  });
  root.querySelectorAll("[data-ev='kpi']").forEach(el => {
    const onEdit = () => {
      if (!el.isConnected) return;
      evalRec(el.dataset.code, el.dataset.year).kpi[el.dataset.kpi] = el.value;
      const k = kpiByCode[el.dataset.kpi];
      const tgt = (k && k.targets) ? k.targets[el.dataset.year] : null;
      const r = evalRate(el.value, tgt);
      const cell = el.closest("tr").querySelector(".ev-rate");
      if (cell) {
        cell.textContent = r === null ? "–" : fmt1(r) + "%";
        cell.classList.toggle("hit", r !== null && r >= 100);
      }
      const mirror = el.parentElement.querySelector(".ev-print");
      if (mirror) mirror.textContent = el.value;
      refreshEvalMeta(el.dataset.code);
    };
    el.addEventListener("input", onEdit);
    el.addEventListener("change", onEdit);
  });
  root.querySelectorAll(".evcard.open textarea").forEach(autoGrow);
  return list.length;
}
function refreshEvalMeta(code) {
  const card = document.querySelector(`#view-eval .evcard[data-code="${code}"]`);
  if (card) {
    const p = card.querySelector(".evprog");
    if (p) {
      const st = evalStatus(code);
      p.textContent = `${evalFilled(code)}/${EVAL_FIELD_TOTAL} 작성`;
      p.className = "evprog" + (st === "done" ? " p-done" : (st === "part" ? " p-part" : ""));
    }
  }
  const bar = document.getElementById("evsum");
  if (bar) bar.innerHTML = evalSummaryInner(DATA.tasks.filter(taskMatch));
}

/* ── tooltip ── */
const tip = document.getElementById("tip");
function showTip(html, ev) {
  tip.innerHTML = html; tip.style.display = "block";
  const pad = 14;
  let x = ev.clientX + pad, y = ev.clientY + pad;
  const r = tip.getBoundingClientRect();
  if (x + r.width > innerWidth - 8) x = ev.clientX - r.width - pad;
  if (y + r.height > innerHeight - 8) y = ev.clientY - r.height - pad;
  tip.style.left = x + "px"; tip.style.top = y + "px";
}
function hideTip() { tip.style.display = "none"; }
document.addEventListener("mousemove", ev => {
  const el = ev.target;
  if (el.classList && el.classList.contains("pt")) {
    const k = kpiByCode[el.dataset.k];
    if (!k) return hideTip();
    const rows = YEARS.map(y => {
      const v = k.targets ? k.targets[y] : null;
      const mark = y === el.dataset.y ? " ◀" : "";
      return `${y}: ${fmt(v)}${v !== null && v !== undefined ? (k.unit || "") : ""}${mark}`;
    }).join("<br>");
    showTip(`<b>${esc(k.code)} ${esc(k.name)}</b><br>기준값 ${fmt(k.baseline)}${k.unit ? esc(k.unit) : ""}${k.baseline_ref ? " (" + esc(k.baseline_ref) + ")" : ""}<div class="rows">${rows}</div>`, ev);
  } else if (el.dataset && el.dataset.tip) {
    showTip(esc(el.dataset.tip), ev);
  } else hideTip();
});

function renderAll() {
  let n = 0, label = "";
  if (state.view === "strategy") { n = renderStrategy(); label = `실행과제 ${n}건 표시`; }
  else if (state.view === "kpi") { n = renderKpi(); label = `KPI ${n}개 표시`; }
  else if (state.view === "compare") { n = renderCompare(); label = `주요지표 ${n}개 비교 (대학알리미)`; }
  else if (state.view === "eval") { n = renderEval(); label = `자체평가 대상 실행과제 ${n}건 (${state.evalYear}학년도)`; }
  else { n = renderDept(); label = `책임부서 ${n}개 표시`; }
  document.getElementById("evyearwrap").classList.toggle("hidden", state.view !== "eval");
  const f = [];
  if (state.goal) { const g = GOALS.find(x => x.id === state.goal); f.push(`전략 ${state.goal}. ${g ? g.name : ""}`); }
  if (state.spec) f.push("특성화 연계만");
  if (state.dept) f.push(`책임부서 = ${state.dept}`);
  if (state.q) f.push(`검색 "${state.q}"`);
  document.getElementById("countnote").textContent = label + (f.length ? " · 필터: " + f.join(" / ") : " · 전체");
}
